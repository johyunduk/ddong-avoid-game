```sql
  id         INTEGER PRIMARY KEY,   -- 0=프리시즌, 1=2026-04, ...
  name       TEXT    NOT NULL,
  starts_at  TIMESTAMPTZ NOT NULL,
  ends_at    TIMESTAMPTZ,           -- NULL = 현재 진행 중

);

INSERT INTO public.seasons (id, name, starts_at) VALUES
  (0, '프리시즌',        '2000-01-01T00:00:00Z'),
  (1, '2026년 4월 시즌', '2026-04-01T00:00:00+09:00');

CREATE OR REPLACE FUNCTION get_current_season_id()
RETURNS INTEGER LANGUAGE sql STABLE AS $$
  SELECT id FROM public.seasons
  WHERE starts_at <= NOW()
    AND (ends_at IS NULL OR ends_at > NOW())
  ORDER BY id DESC LIMIT 1;
$$;
GRANT EXECUTE ON FUNCTION get_current_season_id TO anon, authenticated, service_role;
```

```sql
CREATE OR REPLACE VIEW public.battle_leaderboard_view AS

       br.wins, br.losses, br.disconnects,
```

### Step 4 — `submit_battle_record` 함수 재정의
파일: `supabase/migrations/20260324_submit_battle_record_v3.sql`

```sql
DROP FUNCTION IF EXISTS submit_battle_record(UUID, TEXT, INTEGER, BOOLEAN);

CREATE OR REPLACE FUNCTION submit_battle_record(
  p_user_id     UUID,
  p_result      TEXT,
  p_point_delta INTEGER,
  p_is_ranked   BOOLEAN  DEFAULT TRUE,
  p_season_id   INTEGER  DEFAULT NULL  -- NULL이면 get_current_season_id()
)
RETURNS TABLE(
  out_rating_points        INTEGER,
  out_wins                 INTEGER,
  out_losses               INTEGER,
  out_disconnects          INTEGER,
  out_friendly_wins        INTEGER,
  out_friendly_losses      INTEGER,
  out_friendly_disconnects INTEGER,
  out_rank                 BIGINT
)
LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_sid INTEGER := COALESCE(p_season_id, get_current_season_id());
  v_rp INT; v_w INT; v_l INT; v_d INT;
  v_fw INT; v_fl INT; v_fd INT; v_rank BIGINT;
BEGIN
  IF p_is_ranked THEN
    INSERT INTO public.battle_records (user_id, season_id, rating_points)
    VALUES (p_user_id, v_sid, 0) ON CONFLICT (user_id, season_id) DO NOTHING;

    SELECT br.rating_points, br.wins, br.losses, br.disconnects
      INTO v_rp, v_w, v_l, v_d
      FROM public.battle_records br WHERE br.user_id=p_user_id AND br.season_id=v_sid;

    v_rp := GREATEST(0, COALESCE(v_rp,0) + p_point_delta);
    v_w  := COALESCE(v_w,0) + CASE WHEN p_result='win'        THEN 1 ELSE 0 END;
    v_l  := COALESCE(v_l,0) + CASE WHEN p_result='lose'       THEN 1 ELSE 0 END;
    v_d  := COALESCE(v_d,0) + CASE WHEN p_result='disconnect' THEN 1 ELSE 0 END;

    UPDATE public.battle_records SET
      rating_points=v_rp, wins=v_w, losses=v_l, disconnects=v_d, updated_at=NOW()
    WHERE user_id=p_user_id AND season_id=v_sid;

    SELECT COUNT(*)+1 INTO v_rank FROM public.battle_records br2
    WHERE br2.season_id=v_sid AND br2.rating_points > v_rp;

    SELECT bfr.friendly_wins, bfr.friendly_losses, bfr.friendly_disconnects
      INTO v_fw, v_fl, v_fd
      FROM public.battle_friendly_records bfr WHERE bfr.user_id=p_user_id;
    v_fw:=COALESCE(v_fw,0); v_fl:=COALESCE(v_fl,0); v_fd:=COALESCE(v_fd,0);
  ELSE
    INSERT INTO public.battle_friendly_records (user_id) VALUES (p_user_id)
    ON CONFLICT (user_id) DO NOTHING;

    SELECT bfr.friendly_wins, bfr.friendly_losses, bfr.friendly_disconnects
      INTO v_fw, v_fl, v_fd
      FROM public.battle_friendly_records bfr WHERE bfr.user_id=p_user_id;

    v_fw:=COALESCE(v_fw,0)+CASE WHEN p_result='win'        THEN 1 ELSE 0 END;
    v_fl:=COALESCE(v_fl,0)+CASE WHEN p_result='lose'       THEN 1 ELSE 0 END;
    v_fd:=COALESCE(v_fd,0)+CASE WHEN p_result='disconnect' THEN 1 ELSE 0 END;

    UPDATE public.battle_friendly_records SET
      friendly_wins=v_fw, friendly_losses=v_fl,
      friendly_disconnects=v_fd, updated_at=NOW()
    WHERE user_id=p_user_id;

    SELECT COALESCE(br.rating_points,0), COALESCE(br.wins,0),
           COALESCE(br.losses,0), COALESCE(br.disconnects,0)
      INTO v_rp, v_w, v_l, v_d
      FROM public.battle_records br WHERE br.user_id=p_user_id AND br.season_id=v_sid;
    v_rp:=COALESCE(v_rp,0);

    SELECT COUNT(*)+1 INTO v_rank FROM public.battle_records br2
    WHERE br2.season_id=v_sid AND br2.rating_points > v_rp;
  END IF;

  RETURN QUERY SELECT v_rp,v_w,v_l,v_d,v_fw,v_fl,v_fd,v_rank;
END; $$;

REVOKE ALL ON FUNCTION submit_battle_record FROM PUBLIC;
GRANT EXECUTE ON FUNCTION submit_battle_record TO service_role;

-- get_battle_rank_by_rp에 season_id 파라미터 추가
DROP FUNCTION IF EXISTS get_battle_rank_by_rp(INTEGER);
CREATE OR REPLACE FUNCTION get_battle_rank_by_rp(p_rp INTEGER, p_season_id INTEGER DEFAULT NULL)
RETURNS BIGINT LANGUAGE sql STABLE AS $$
  SELECT COUNT(*) FROM public.battle_records
  WHERE season_id = COALESCE(p_season_id, get_current_season_id())
    AND rating_points > p_rp;
$$;
GRANT EXECUTE ON FUNCTION get_battle_rank_by_rp TO anon, authenticated, service_role;
```

```typescript
// user_skor에서 현재 balance를 IN 쿼리로 일괄 조회 후 upsert
async function bulkAddSkor(admin, rewardMap: Record<string, number>) {
  const userIds = Object.keys(rewardMap);
  const { data: existing } = await admin.from('user_skor').select('user_id,balance').in('user_id', userIds);
  const balanceMap = Object.fromEntries((existing ?? []).map(r => [r.user_id, r.balance]));
  await admin.from('user_skor').upsert(
    userIds.map(uid => ({ user_id: uid, balance: Math.floor((balanceMap[uid] ?? 0) + rewardMap[uid]), updated_at: new Date().toISOString() })),
    { onConflict: 'user_id' }
  );
}
```
