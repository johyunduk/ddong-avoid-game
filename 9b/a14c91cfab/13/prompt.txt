## 1. 개요

스토리는 **100% 텍스트만**으로 전달한다.  
강제 노출 없음 — 홈 화면 버튼을 통해 플레이어가 원할 때만 확인.

```
홈 화면
  └── 📁 기억 로그 버튼 (새 로그 있을 때 빨간 점 표시)
        └── StoryLogScene (아카이브 리스트)
              └── 로그 뷰어 (터미널 스타일 텍스트)
```

## 5. 해금 진행 관리 (`src/utils/storyProgress.ts`)

```typescript
import { STORY_LOGS } from '../data/storyLogs';
import type { StoryLog } from '../types/StoryTypes';

// localStorage 키
const UNLOCKED_KEY  = 'storyUnlocked';   // string[]  — 해금된 로그 ID 배열
const READ_KEY      = 'storyRead';        // string[]  — 읽은 로그 ID 배열
const STATS_KEY     = 'storyStats';       // StoryStats JSON

interface StoryStats {
  topazCollected:   number;
  goldCollected:    number;
  diamondCollected: number;
  playCount:        number;
  totalSkor:        number;
  gachaPullCount:   number;
}

function getStats(): StoryStats {
  try {
    return JSON.parse(localStorage.getItem(STATS_KEY) || '{}');
  } catch { return {} as StoryStats; }
}

function saveStats(stats: StoryStats): void {
  localStorage.setItem(STATS_KEY, JSON.stringify(stats));
}

/** 게임 이벤트 발생 시 통계 누적 및 해금 체크 */
export function recordEvent(
  type: 'topaz' | 'gold' | 'diamond' | 'playCount' | 'skor' | 'gacha',
  amount = 1
): void {
  const stats = getStats();
  switch (type) {
    case 'topaz':     stats.topazCollected   = (stats.topazCollected   || 0) + amount; break;
    case 'gold':      stats.goldCollected    = (stats.goldCollected    || 0) + amount; break;
    case 'diamond':   stats.diamondCollected = (stats.diamondCollected || 0) + amount; break;
    case 'playCount': stats.playCount        = (stats.playCount        || 0) + amount; break;
    case 'skor':      stats.totalSkor        = (stats.totalSkor        || 0) + amount; break;
    case 'gacha':     stats.gachaPullCount   = (stats.gachaPullCount   || 0) + amount; break;
  }
  saveStats(stats);
  checkAndUnlockLogs(stats);
}

function checkAndUnlockLogs(stats: StoryStats): void {
  const unlocked = getUnlockedIds();
  let changed = false;

  for (const log of STORY_LOGS) {
    if (unlocked.includes(log.id)) continue;

    const { type, threshold } = log.unlockCondition;
    const current =
      type === 'topaz'     ? (stats.topazCollected   || 0) :
      type === 'gold'      ? (stats.goldCollected    || 0) :
      type === 'diamond'   ? (stats.diamondCollected || 0) :
      type === 'playCount' ? (stats.playCount        || 0) :
      type === 'skor'      ? (stats.totalSkor        || 0) :
      type === 'gacha'     ? (stats.gachaPullCount   || 0) : 0;

    if (current >= threshold) {
      unlocked.push(log.id);
      changed = true;
      // 해금 알림 (강제 팝업 없이 — 아카이브 화면 진입 시 표시)
      console.log(`[Story] 새 로그 해금: ${log.id}`);
    }
  }

  if (changed) {
    localStorage.setItem(UNLOCKED_KEY, JSON.stringify(unlocked));
  }
}

export function getUnlockedIds(): string[] {
  try { return JSON.parse(localStorage.getItem(UNLOCKED_KEY) || '[]'); }
  catch { return []; }
}

export function getReadIds(): string[] {
  try { return JSON.parse(localStorage.getItem(READ_KEY) || '[]'); }
  catch { return []; }
}

export function markAsRead(id: string): void {
  const read = getReadIds();
  if (!read.includes(id)) {
    read.push(id);
    localStorage.setItem(READ_KEY, JSON.stringify(read));
  }
}

/** 홈 화면 빨간 점 표시 여부 */
export function hasUnreadNewLogs(): boolean {
  const unlocked = getUnlockedIds();
  const read = getReadIds();
  return unlocked.some(id => !read.includes(id));
}
```

```typescript
import Phaser from 'phaser';
import { STORY_LOGS } from '../data/storyLogs';
import {
  getUnlockedIds,
  getReadIds,
  markAsRead,
} from '../utils/storyProgress';
import BaseScene from './BaseScene';

const SCROLL_TOP    = 95;
const SCROLL_BOTTOM = 548;
const ROW_H         = 72;
const ROW_GAP       = 6;

export default class StoryLogScene extends BaseScene {
  private scrollOffset = 0;
  private maxScrollOffset = 0;
  private rowContainer!: Phaser.GameObjects.Container;
  private maskGfx!: Phaser.GameObjects.Graphics;
  private pointerDownY = 0;
  private pointerDownScrollY = 0;
  private hasDragged = false;

  constructor() { super('StoryLogScene'); }

  create() {
    super.create();

    const W = this.scale.width;
    const H = this.scale.height;
    const cx = W / 2;
    const yOff = (H - 600) / 2;

    // 배경
    this.add.rectangle(cx, H / 2, W, H, 0x080812);

    // 헤더
    this.drawHeader(cx, yOff);

    // 스크롤 가능한 로그 리스트
    this.rowContainer = this.add.container(0, yOff);
    this.buildLogList();

    // 마스크 (스크롤 영역 밖 숨김)
    this.maskGfx = this.make.graphics({ x: 0, y: 0 });
    this.maskGfx.fillStyle(0xffffff);
    this.maskGfx.fillRect(
      0,
      SCROLL_TOP + yOff,
      W,
      H - (600 - SCROLL_BOTTOM) - SCROLL_TOP - yOff,
    );
    this.rowContainer.setMask(this.maskGfx.createGeometryMask());

    // 스크롤 입력
    this.setupScrollInput(yOff, H);

    // 뒤로가기
    this.createBackButton(cx, H - 27, yOff);

    this.events.once('shutdown', () => this.maskGfx.destroy());
  }

  private drawHeader(cx: number, yOff: number) {
    // 헤더 배경
    this.add.rectangle(cx, 40 + yOff, this.scale.width, 48, 0x000000, 0.8);
    this.add.rectangle(cx, 64 + yOff, this.scale.width, 1, 0x00ff41, 0.2);

    // 제목
    this.add.text(cx, 34 + yOff, '기억 로그 아카이브', {
      fontSize: '14px',
      color: '#00ff41',
      fontStyle: 'bold',
      stroke: '#000',
      strokeThickness: 3,
    }).setOrigin(0.5);

    // 새 로그 수 뱃지
    const unread = getUnlockedIds().filter(id => !getReadIds().includes(id));
    if (unread.length > 0) {
      const badge = this.add.circle(cx + 80, 28 + yOff, 10, 0xff4444);
      this.add.text(cx + 80, 28 + yOff, String(unread.length), {
        fontSize: '10px', color: '#fff', fontStyle: 'bold',
      }).setOrigin(0.5).setDepth(badge.depth + 1);
    }
  }

  private buildLogList() {
    const W = this.scale.width;
    const cx = W / 2;
    const unlocked = getUnlockedIds();
    const read = getReadIds();

    let y = SCROLL_TOP + 10;
    let currentSeason = 0;

    for (const log of STORY_LOGS) {
      // 시즌 구분선
      if (log.season !== currentSeason) {
        currentSeason = log.season;
        const label = this.add.text(16, y, `— SEASON ${currentSeason} —`, {
          fontSize: '9px', color: '#00ff4166',
          fontFamily: 'monospace',
        });
        this.rowContainer.add(label);
        y += 24;
      }

      const isUnlocked = unlocked.includes(log.id);
      const isRead = read.includes(log.id);
      const isNew = isUnlocked && !isRead;

      this.createLogRow(cx, y, log, isUnlocked, isNew);
      y += ROW_H + ROW_GAP;
    }

    // 최대 스크롤 계산
    const scrollBottom = this.scale.height - (600 - SCROLL_BOTTOM);
    this.maxScrollOffset = Math.max(
      0,
      this.yOff + y - scrollBottom,
    );
  }

  private createLogRow(
    cx: number,
    y: number,
    log: (typeof STORY_LOGS)[0],
    isUnlocked: boolean,
    isNew: boolean,
  ) {
    const W = this.scale.width;
    const alpha = isUnlocked ? 1 : 0.4;

    // 카드 배경
    const bg = this.add.rectangle(cx, y + ROW_H / 2, W - 24, ROW_H, 0x0d0d1e, 1)
      .setStrokeStyle(1, isNew ? 0x00ff4144 : 0xffffff15)
      .setAlpha(alpha);
    this.rowContainer.add(bg);

    // 아이콘
    const icon = this.add.text(24, y + ROW_H / 2, isUnlocked ? (isNew ? '📋' : '📄') : '🔒', {
      fontSize: '18px',
    }).setOrigin(0, 0.5).setAlpha(alpha);
    this.rowContainer.add(icon);

    // 로그 ID
    const idColor = isUnlocked ? '#00ff41' : '#444444';
    const idText = this.add.text(48, y + ROW_H / 2 - 12, `[${log.id}]`, {
      fontSize: '9px', color: idColor, fontFamily: 'monospace',
    }).setAlpha(alpha);
    this.rowContainer.add(idText);

    // NEW 뱃지
    if (isNew) {
      const newBg = this.add.rectangle(
        48 + idText.width + 18, y + ROW_H / 2 - 12,
        28, 14, 0xff4444,
      );
      this.add.text(
        newBg.x, newBg.y, 'NEW',
        { fontSize: '8px', color: '#fff', fontStyle: 'bold' },
      ).setOrigin(0.5);
      this.rowContainer.add(newBg);
    }

    // 제목 or ?????
    this.rowContainer.add(
      this.add.text(48, y + ROW_H / 2 + 2,
        isUnlocked ? log.title : '?????', {
          fontSize: '13px',
          color: isUnlocked ? '#cccccc' : '#333333',
          fontStyle: 'bold',
          fontFamily: 'Noto Sans KR, sans-serif',
        },
      ).setAlpha(alpha)
    );

    // 잠금 조건 (미해금)
    if (!isUnlocked) {
      this.rowContainer.add(
        this.add.text(48, y + ROW_H / 2 + 20,
          `해금 조건: ${log.unlockCondition.type} × ${log.unlockCondition.threshold}`, {
            fontSize: '10px', color: '#555555',
            fontFamily: 'Noto Sans KR, sans-serif',
          }
        )
      );
    }

    // 화살표 (해금된 항목만)
    if (isUnlocked) {
      this.rowContainer.add(
        this.add.text(W - 20, y + ROW_H / 2, '▶', {
          fontSize: '12px', color: '#00ff4155',
        }).setOrigin(1, 0.5)
      );
    }

    // 클릭 이벤트
    if (isUnlocked) {
      bg.setInteractive({ useHandCursor: true });
      bg.on('pointerover', () => bg.setFillStyle(0x00ff4108));
      bg.on('pointerout',  () => bg.setFillStyle(0x0d0d1e));
      bg.on('pointerup',   () => {
        if (!this.hasDragged) this.openLogViewer(log);
      });
    }
  }

  private openLogViewer(log: (typeof STORY_LOGS)[0]) {
    markAsRead(log.id);
    // StoryLogViewerScene을 overlay로 실행
    this.scene.launch('StoryLogViewerScene', { log });
    this.scene.pause();
  }

  private setupScrollInput(yOff: number, H: number) {
    const scrollTopActual    = SCROLL_TOP + yOff;
    const scrollBottomActual = H - (600 - SCROLL_BOTTOM);

    this.input.on('pointerdown', (p: Phaser.Input.Pointer) => {
      if (p.y < scrollTopActual || p.y > scrollBottomActual) return;
      this.pointerDownY = p.y;
      this.pointerDownScrollY = this.scrollOffset;
      this.hasDragged = false;
    });
    this.input.on('pointermove', (p: Phaser.Input.Pointer) => {
      if (!p.isDown) return;
      const dy = this.pointerDownY - p.y;
      if (Math.abs(dy) > 5) {
        this.hasDragged = true;
        this.scrollOffset = Phaser.Math.Clamp(
          this.pointerDownScrollY + dy, 0, this.maxScrollOffset,
        );
        this.rowContainer.setY(this.yOff - this.scrollOffset);
      }
    });
    this.input.on('pointerup', () => {
      this.time.delayedCall(0, () => { this.hasDragged = false; });
    });
    this.input.on('wheel', (_p: unknown, _go: unknown, _dx: number, dy: number) => {
      this.scrollOffset = Phaser.Math.Clamp(
        this.scrollOffset + dy * 0.5, 0, this.maxScrollOffset,
      );
      this.rowContainer.setY(this.yOff - this.scrollOffset);
    });
  }

  private createBackButton(cx: number, y: number, yOff: number) {
    const btn = this.add.rectangle(cx, y, 200, 50, 0x333333)
      .setStrokeStyle(1, 0x444444)
      .setDepth(10)
      .setInteractive({ useHandCursor: true });
    this.add.text(cx, y, '← 돌아가기', {
      fontSize: '16px', color: '#ffffff', fontStyle: 'bold',
    }).setOrigin(0.5).setDepth(11);
    btn.on('pointerover', () => btn.setFillStyle(0x555555));
    btn.on('pointerout',  () => btn.setFillStyle(0x333333));
    btn.on('pointerdown', () => this.scene.start('ModeSelectScene'));
  }

  // yOff getter (BaseScene에서 상속 못 할 경우 직접 계산)
  private get yOff(): number {
    return (this.scale.height - 600) / 2;
  }
}
```

# Obsidian CLI

## Syntax Rules

```bash
obsidian <command> [key=value ...]
```

- `file=<name>` — 노트 이름으로 해석 (위키링크처럼)
- `path=<path>` — 정확한 경로로 해석 (예: `folder/note.md`)
- file/path 생략 시 **현재 활성 파일** 기본값
- `vault=<name>` — 특정 vault 지정 (생략 시 활성 vault)
- 공백 포함 값은 따옴표 사용: `name="My Note"`
- 내용 내 줄바꿈: `\n`, 탭: `\t`

## 자주 쓰는 패턴

### 노트 읽기 / 생성 / 편집
```bash
obsidian read file="Note Name"
obsidian read path="folder/note.md"

obsidian create name="New Note" content="첫 내용"
obsidian create name="New Note" template="My Template" open

obsidian append file="Note Name" content="추가할 내용"
obsidian prepend file="Note Name" content="앞에 붙일 내용\n"
```

### Daily Note
```bash
obsidian daily:read
obsidian daily:append content="- [ ] 오늘 할 일"
obsidian daily:path
```

### 검색
```bash
obsidian search query="검색어"
obsidian search query="키워드" path="folder" limit=10
obsidian search:context query="검색어"   # 매칭 라인 컨텍스트 포함
```

### Tasks
```bash
obsidian tasks todo                          # 미완료 작업 목록
obsidian tasks done file="Note Name"         # 특정 파일의 완료 작업
obsidian task toggle path="folder/note.md" line=5
obsidian task done daily                     # 데일리 노트 작업 완료 처리
```

### Properties (Frontmatter)
```bash
obsidian property:read name="status" file="Note Name"
obsidian property:set name="status" value="done" file="Note Name"
obsidian property:set name="tags" value="project, active" type=list file="Note Name"
obsidian property:remove name="draft" file="Note Name"
```

### Vault 구조 점검
```bash
obsidian orphans          # 인커밍 링크 없는 파일
obsidian deadends         # 아웃고잉 링크 없는 파일
obsidian unresolved       # 깨진 위키링크
obsidian backlinks file="Note Name"
```

## 유용한 플래그

| 플래그 | 용도 |
|--------|------|
| `total` | 개수만 반환: `obsidian files total` |
| `verbose` | 상세 정보 추가 |
| `format=json` | JSON 출력: `obsidian search query="x" format=json` |
| `format=csv\|tsv` | CSV/TSV 출력 |
